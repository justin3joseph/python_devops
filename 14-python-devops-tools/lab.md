    data = yaml.safe_load(file)

        print(server["name"], "valid")
    except ValueError:
        print(server["name"], "invalid")

with open("servers.yaml") as file:
    data = yaml.safe_load(file)

with open("inventory.ini", "w") as file:
    file.write("[web]\n")
    for server in data["servers"]:
        file.write(
            f'{server["name"]} ansible_host={server["ip"]}\n'
        )

print("Ansible inventory generated.")
